-- =========================
-- BASIC SQL
-- =========================
SELECT * FROM investors;

-- =========================
-- INTERMEDIATE SQL
-- =========================
SELECT * FROM transactions WHERE quantity > 0;

-- =========================
-- COMPLEX QUERY (JOIN)
-- =========================
SELECT 
    i.name,
    a.name,
    t.quantity,
    t.price
FROM transactions t
JOIN investors i ON t.investor_id = i.investor_id
JOIN assets a ON t.asset_id = a.asset_id;

-- =========================
-- AGGREGATION
-- =========================
SELECT 
    asset_id,
    SUM(quantity)
FROM transactions
GROUP BY asset_id;
